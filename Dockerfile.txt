# 1. Start with an official lightweight Python setup
FROM python:3.10-slim

# 2. Set the working folder inside the container
WORKDIR /app

# 3. Copy your project's list of dependencies into the container
COPY requirements.txt .

# 4. Install those dependencies inside the container
RUN pip install --no-cache-dir -r requirements.txt

# 5. Copy the rest of your app's code into the container
COPY . .

# 6. Specify the command to run your app
CMD ["python", "app.py"]
